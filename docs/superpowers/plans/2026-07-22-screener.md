# Long/Short + Narrative Screener Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `/screener` page with two independent panels — a sortable Bybit-backed long/short market-positioning table (top 3 long-leaning / top 3 short-leaning pairs flagged) and a CoinGecko-backed trending watchlist as a narrative-forming signal.

**Architecture:** Two new backend services (`services/bybit.ts`, `services/coingecko.ts`) behind two new routes (`GET /api/screener/majors`, `GET /api/screener/trending`), each independently cached and independently fault-tolerant. New frontend page renders both panels from two separate loading/error/ready states so one failing never blanks the other.

**Tech Stack:** Same as the existing backend (Fastify, TypeScript, Vitest) and frontend (React, TypeScript, Tailwind, Vitest) — no new dependencies required; both external APIs are called with the built-in `fetch`.

## Global Constraints

- Binance's API is blocked from the user's network — **do not use Binance for anything in this module.** Bybit was confirmed reachable and is the majors data source.
- Both new backend services must be dependency-injectable (`fetchImpl`, `now`, `cacheTtlMs`) the same way `services/rss.ts` and `services/cryptopanic`-successor already are in this codebase, so tests never hit the real network.
- `/api/screener/majors` and `/api/screener/trending` are separate endpoints — never merge them into one response. A CoinGecko failure must never affect the Bybit-backed panel or vice versa.
- Fixed pair list for majors (exact 15, exact order, exact Bybit symbol spelling): `BTCUSDT, ETHUSDT, SOLUSDT, BNBUSDT, XRPUSDT, DOGEUSDT, ADAUSDT, AVAXUSDT, LINKUSDT, TONUSDT, SUIUSDT, DOTUSDT, LTCUSDT, TRXUSDT, NEARUSDT`.
- Cache TTLs: majors ~5 minutes (matches Bybit's own `5min` account-ratio granularity), trending ~10 minutes (CoinGecko free tier has a tighter rate limit).
- The trending panel must visibly disclose it is a search-interest signal, not investment advice — this is a copy requirement, not just a design preference.
- Follow existing code style: function components, inline `style={{ color: 'var(--...)' }}` for design tokens, reuse `formatPercent` from `frontend/src/lib/format.ts` rather than reimplementing percentage formatting.

// backend/src/services/bybit.ts
export const PAIR_LIST = [
  'BTCUSDT', 'ETHUSDT', 'SOLUSDT', 'BNBUSDT', 'XRPUSDT',
  'DOGEUSDT', 'ADAUSDT', 'AVAXUSDT', 'LINKUSDT', 'TONUSDT',
  'SUIUSDT', 'DOTUSDT', 'LTCUSDT', 'TRXUSDT', 'NEARUSDT',
];

export interface PairPositioning {
  symbol: string;
  buyRatio: number;
  sellRatio: number;
  fundingRate: number;
  price24hPct: number;
}

export interface MajorsResult {
  pairs: PairPositioning[];
  topLongSymbols: string[];
  topShortSymbols: string[];
}

export class BybitUnavailableError extends Error {}

export interface BybitServiceDeps {
  fetchImpl?: typeof fetch;
  now?: () => number;
  cacheTtlMs?: number;
}

const DEFAULT_CACHE_TTL_MS = 5 * 60 * 1000;
const TICKERS_URL = 'https://api.bybit.com/v5/market/tickers?category=linear';

function accountRatioUrl(symbol: string): string {
  return `https://api.bybit.com/v5/market/account-ratio?category=linear&symbol=${symbol}&period=5min&limit=1`;
}

async function fetchBulkTickers(fetchImpl: typeof fetch): Promise<Map<string, { fundingRate: number; price24hPct: number }>> {
  let response: Response;
  try {
    response = await fetchImpl(TICKERS_URL);
  } catch (err) {
    throw new BybitUnavailableError('Failed to reach Bybit tickers endpoint', { cause: err });
  }
  if (!response.ok) {
    throw new BybitUnavailableError(`Bybit tickers endpoint returned status ${response.status}`);
  }

  let payload: unknown;
  try {
    payload = await response.json();
  } catch (err) {
    throw new BybitUnavailableError('Bybit tickers response was not valid JSON', { cause: err });
  }

  const list = (payload as { result?: { list?: unknown } })?.result?.list;
  if (!Array.isArray(list)) {
    throw new BybitUnavailableError('Bybit tickers response missing expected fields');
  }

  const map = new Map<string, { fundingRate: number; price24hPct: number }>();
  for (const raw of list) {
    const item = raw as Record<string, unknown>;
    if (typeof item.symbol !== 'string') continue;
    const fundingRate = Number(item.fundingRate);
    const price24hPcnt = Number(item.price24hPcnt);
    if (Number.isNaN(fundingRate) || Number.isNaN(price24hPcnt)) continue;
    map.set(item.symbol, { fundingRate, price24hPct: price24hPcnt * 100 });
  }
  return map;
}

async function fetchAccountRatio(
  symbol: string,
  fetchImpl: typeof fetch,
): Promise<{ symbol: string; buyRatio: number; sellRatio: number } | null> {
  let response: Response;
  try {
    response = await fetchImpl(accountRatioUrl(symbol));
  } catch {
    return null;
  }
  if (!response.ok) return null;

  let payload: unknown;
  try {
    payload = await response.json();
  } catch {
    return null;
  }

  const list = (payload as { result?: { list?: unknown } })?.result?.list;
  if (!Array.isArray(list) || list.length === 0) return null;

  const entry = list[0] as Record<string, unknown>;
  const buyRatio = Number(entry.buyRatio);
  const sellRatio = Number(entry.sellRatio);
  if (Number.isNaN(buyRatio) || Number.isNaN(sellRatio)) return null;

  return { symbol, buyRatio, sellRatio };
}

export function createBybitService(deps: BybitServiceDeps = {}) {
  const fetchImpl = deps.fetchImpl ?? fetch;
  const now = deps.now ?? Date.now;
  const cacheTtlMs = deps.cacheTtlMs ?? DEFAULT_CACHE_TTL_MS;
  let cache: { expiresAt: number; data: MajorsResult } | null = null;

  async function getMajors(): Promise<MajorsResult> {
    if (cache && cache.expiresAt > now()) {
      return cache.data;
    }

    const tickerMap = await fetchBulkTickers(fetchImpl);
    const ratioResults = await Promise.all(PAIR_LIST.map((symbol) => fetchAccountRatio(symbol, fetchImpl)));

    const pairs: PairPositioning[] = [];
    for (const ratio of ratioResults) {
      if (!ratio) continue;
      const ticker = tickerMap.get(ratio.symbol);
      if (!ticker) continue;
      pairs.push({
        symbol: ratio.symbol,
        buyRatio: ratio.buyRatio,
        sellRatio: ratio.sellRatio,
        fundingRate: ticker.fundingRate,
        price24hPct: ticker.price24hPct,
      });
    }

    const sortedByBuyRatioDesc = [...pairs].sort((a, b) => b.buyRatio - a.buyRatio);
    const topLongSymbols = sortedByBuyRatioDesc.slice(0, 3).map((p) => p.symbol);
    const topShortSymbols = sortedByBuyRatioDesc.slice(-3).reverse().map((p) => p.symbol);

    const result: MajorsResult = { pairs, topLongSymbols, topShortSymbols };
    cache = { expiresAt: now() + cacheTtlMs, data: result };
    return result;
  }

  return { getMajors };
}

// backend/src/services/coingecko.ts
export interface TrendingCoin {
  id: string;
  name: string;
  symbol: string;
  marketCapRank: number | null;
  priceUsd: number;
  priceChangePct24h: number;
}

export class CoinGeckoUnavailableError extends Error {}

export interface CoinGeckoServiceDeps {
  fetchImpl?: typeof fetch;
  now?: () => number;
  cacheTtlMs?: number;
}

const DEFAULT_CACHE_TTL_MS = 10 * 60 * 1000;
const TRENDING_URL = 'https://api.coingecko.com/api/v3/search/trending';

function parseTrending(payload: unknown): TrendingCoin[] | null {
  if (typeof payload !== 'object' || payload === null || !('coins' in payload)) return null;
  const coins = (payload as { coins: unknown }).coins;
  if (!Array.isArray(coins)) return null;

  const result: TrendingCoin[] = [];
  for (const raw of coins) {
    const item = (raw as Record<string, unknown> | undefined)?.item as Record<string, unknown> | undefined;
    if (!item) return null;
    const data = item.data as Record<string, unknown> | undefined;
    const priceChangeObj = data?.price_change_percentage_24h as Record<string, unknown> | undefined;
    const priceChange = priceChangeObj?.usd;

    if (
      typeof item.id !== 'string' ||
      typeof item.name !== 'string' ||
      typeof item.symbol !== 'string' ||
      typeof data?.price !== 'number' ||
      typeof priceChange !== 'number'
    ) {
      return null;
    }

    result.push({
      id: item.id,
      name: item.name,
      symbol: item.symbol,
      marketCapRank: typeof item.market_cap_rank === 'number' ? item.market_cap_rank : null,
      priceUsd: data.price,
      priceChangePct24h: priceChange,
    });
  }
  return result;
}

function sortByMarketCapRankDesc(coins: TrendingCoin[]): TrendingCoin[] {
  return [...coins].sort((a, b) => {
    if (a.marketCapRank === null && b.marketCapRank === null) return 0;
    if (a.marketCapRank === null) return -1;
    if (b.marketCapRank === null) return 1;
    return b.marketCapRank - a.marketCapRank;
  });
}

export function createCoinGeckoService(deps: CoinGeckoServiceDeps = {}) {
  const fetchImpl = deps.fetchImpl ?? fetch;
  const now = deps.now ?? Date.now;
  const cacheTtlMs = deps.cacheTtlMs ?? DEFAULT_CACHE_TTL_MS;
  let cache: { expiresAt: number; data: TrendingCoin[] } | null = null;

  async function getTrending(): Promise<TrendingCoin[]> {
    if (cache && cache.expiresAt > now()) {
      return cache.data;
    }

    let response: Response;
    try {
      response = await fetchImpl(TRENDING_URL);
    } catch (err) {
      throw new CoinGeckoUnavailableError('Failed to reach CoinGecko trending endpoint', { cause: err });
    }

    if (!response.ok) {
      throw new CoinGeckoUnavailableError(`CoinGecko returned status ${response.status}`);
    }

    let payload: unknown;
    try {
      payload = await response.json();
    } catch (err) {
      throw new CoinGeckoUnavailableError('CoinGecko response was not valid JSON', { cause: err });
    }

    const coins = parseTrending(payload);
    if (coins === null) {
      throw new CoinGeckoUnavailableError('CoinGecko response missing expected fields');
    }

    const sorted = sortByMarketCapRankDesc(coins);
    cache = { expiresAt: now() + cacheTtlMs, data: sorted };
    return sorted;
  }

  return { getTrending };
}

// backend/src/routes/screener.ts
import type { FastifyInstance } from 'fastify';
import { createBybitService, BybitUnavailableError, type MajorsResult } from '../services/bybit.js';
import { createCoinGeckoService, CoinGeckoUnavailableError, type TrendingCoin } from '../services/coingecko.js';

interface BybitServiceLike {
  getMajors(): Promise<MajorsResult>;
}
interface CoinGeckoServiceLike {
  getTrending(): Promise<TrendingCoin[]>;
}

export function registerScreenerRoutes(
  app: FastifyInstance,
  deps: { bybitService?: BybitServiceLike; coinGeckoService?: CoinGeckoServiceLike } = {},
): void {
  const bybitService = deps.bybitService ?? createBybitService();
  const coinGeckoService = deps.coinGeckoService ?? createCoinGeckoService();

  app.get('/api/screener/majors', async (_request, reply) => {
    try {
      return await bybitService.getMajors();
    } catch (err) {
      if (err instanceof BybitUnavailableError) {
        reply.code(503);
        return { error: 'Market positioning data is unavailable right now.' };
      }
      throw err;
    }
  });

  app.get('/api/screener/trending', async (_request, reply) => {
    try {
      const coins = await coinGeckoService.getTrending();
      return { coins };
    } catch (err) {
      if (err instanceof CoinGeckoUnavailableError) {
        reply.code(503);
        return { error: 'Trending data is unavailable right now.' };
      }
      throw err;
    }
  });
}

import { useMemo, useState } from 'react';
import type { PairPositioning } from '../types/screener';
import { formatPercent } from '../lib/format';

type SortKey = 'buyRatio' | 'fundingRate' | 'price24hPct';

const columns: { key: SortKey; label: string }[] = [
  { key: 'buyRatio', label: 'Long/Short Ratio' },
  { key: 'fundingRate', label: 'Funding Rate' },
  { key: 'price24hPct', label: '24h Change' },
];

interface PositioningTableProps {
  pairs: PairPositioning[];
  topLongSymbols: string[];
  topShortSymbols: string[];
}

export function PositioningTable({ pairs, topLongSymbols, topShortSymbols }: PositioningTableProps) {
  const [sortKey, setSortKey] = useState<SortKey>('buyRatio');
  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('desc');

  const sorted = useMemo(() => {
    const copy = [...pairs];
    copy.sort((a, b) => {
      const diff = a[sortKey] - b[sortKey];
      return sortDir === 'asc' ? diff : -diff;
    });
    return copy;
  }, [pairs, sortKey, sortDir]);

  function toggleSort(key: SortKey) {
    if (key === sortKey) {
      setSortDir((d) => (d === 'asc' ? 'desc' : 'asc'));
    } else {
      setSortKey(key);
      setSortDir('desc');
    }
  }

  return (
    <div className="overflow-x-auto rounded-lg border" style={{ borderColor: 'var(--border)', background: 'var(--surface-1)' }}>
      <table className="w-full border-collapse text-sm">
        <thead>
          <tr style={{ borderBottom: '1px solid var(--gridline)' }}>
            <th className="px-4 py-3 text-left font-medium" style={{ color: 'var(--text-muted)' }}>
              Pair
            </th>
            {columns.map((col) => (
              <th
                key={col.key}
                className="cursor-pointer select-none px-4 py-3 text-right font-medium"
                style={{ color: 'var(--text-muted)' }}
                onClick={() => toggleSort(col.key)}
              >
                {col.label}
                {sortKey === col.key ? (sortDir === 'asc' ? ' ▲' : ' ▼') : ''}
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {sorted.map((pair) => {
            const isTopLong = topLongSymbols.includes(pair.symbol);
            const isTopShort = topShortSymbols.includes(pair.symbol);
            return (
              <tr key={pair.symbol} style={{ borderBottom: '1px solid var(--gridline)' }}>
                <td className="px-4 py-3 font-medium" style={{ color: 'var(--text-primary)' }}>
                  {pair.symbol}
                  {isTopLong ? (
                    <span className="ml-2 text-xs" style={{ color: 'var(--status-good)' }}>
                      Long-leaning
                    </span>
                  ) : null}
                  {isTopShort ? (
                    <span className="ml-2 text-xs" style={{ color: 'var(--status-critical)' }}>
                      Short-leaning
                    </span>
                  ) : null}
                </td>
                <td className="px-4 py-3 text-right tabular-nums" style={{ color: 'var(--text-primary)' }}>
                  {(pair.buyRatio * 100).toFixed(1)}% / {(pair.sellRatio * 100).toFixed(1)}%
                </td>
                <td className="px-4 py-3 text-right tabular-nums" style={{ color: 'var(--text-primary)' }}>
                  {formatPercent(pair.fundingRate * 100, 4)}
                </td>
                <td
                  className="px-4 py-3 text-right tabular-nums"
                  style={{ color: pair.price24hPct >= 0 ? 'var(--status-good)' : 'var(--status-critical)' }}
                >
                  {formatPercent(pair.price24hPct)}
                </td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
}
