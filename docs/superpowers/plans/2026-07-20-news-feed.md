# News Feed Module Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `/news` page to the dashboard showing a filterable, up-to-date crypto news feed, backed by a new Fastify backend that proxies and caches the CryptoPanic API so the API key never reaches the browser.

**Architecture:** New `backend/` service (Fastify + TypeScript) exposes `GET /api/news`, calls CryptoPanic server-side, caches per query combination in-memory with a short TTL. New frontend page `/news` (in the existing `frontend/` Vite app) calls that endpoint and renders filters + a news list, with explicit loading/empty/error states.

**Tech Stack:** Fastify 5, @fastify/cors, TypeScript (NodeNext modules), tsx (dev runner), Vitest (backend unit tests) — frontend reuses the existing React + TypeScript + Tailwind + react-router-dom stack from `frontend/`.

## Global Constraints

- Backend and frontend are separate npm projects (`backend/`, `frontend/`) — do not merge them or add a root-level package.json.
- The CryptoPanic API key must never be sent to or embedded in frontend code — it only ever lives in `backend/.env` (`CRYPTOPANIC_API_KEY`), read server-side.
- **Requires a free CryptoPanic API key** from https://cryptopanic.com/developers/api/ — the user must obtain this themselves; it cannot be generated by an agent. Task 5's manual verification step needs it; Tasks 1–4 (scaffold + service logic) do not, since their tests mock the upstream call.
- CryptoPanic's exact current query-parameter names and response shape should be spot-checked against https://cryptopanic.com/developers/api/ during Task 2, since third-party API details drift over time — this plan implements against the long-documented `v1/posts/` shape (`auth_token`, `public`, `currencies`, `filter` query params; `results` array of posts with `id`, `title`, `url`, `source`, `published_at`, `currencies`, `votes`). If the live docs differ, adjust `parseCryptoPanicResponse` in Task 2 to match — the rest of the plan (caching, error classification, routing, frontend) is unaffected by exact field names.
- Backend dev server runs on port `4000` by default (`PORT` env var to override) — check it's free before starting, the same way port `5173` was already taken by another local project.
- Follow the existing code style in `frontend/src/` (function components, inline `style={{ color: 'var(--...)' }}` for design-token colors, no CSS-in-JS libraries).

```typescript
// backend/src/services/cryptopanic.ts
export interface CryptoPanicPost {
  id: number;
  title: string;
  url: string;
  sourceTitle: string;
  publishedAt: string;
  currencies: string[];
  votes: {
    positive: number;
    negative: number;
    important: number;
  };
}

export interface NewsQuery {
  currencies?: string;
  filter?: string;
}

export class UpstreamUnavailableError extends Error {}
export class UpstreamRejectedError extends Error {}
export class UpstreamMalformedError extends Error {}

export interface NewsServiceDeps {
  apiKey: string;
  fetchImpl?: typeof fetch;
  now?: () => number;
  cacheTtlMs?: number;
}

const DEFAULT_CACHE_TTL_MS = 4 * 60 * 1000;
const CRYPTOPANIC_POSTS_URL = 'https://cryptopanic.com/api/v1/posts/';

function buildRequestUrl(query: NewsQuery, apiKey: string): string {
  const url = new URL(CRYPTOPANIC_POSTS_URL);
  url.searchParams.set('auth_token', apiKey);
  url.searchParams.set('public', 'true');
  if (query.currencies) url.searchParams.set('currencies', query.currencies);
  if (query.filter) url.searchParams.set('filter', query.filter);
  return url.toString();
}

function parsePosts(payload: unknown): CryptoPanicPost[] | null {
  if (typeof payload !== 'object' || payload === null || !('results' in payload)) return null;
  const results = (payload as { results: unknown }).results;
  if (!Array.isArray(results)) return null;

  const posts: CryptoPanicPost[] = [];
  for (const raw of results) {
    if (typeof raw !== 'object' || raw === null) return null;
    const r = raw as Record<string, unknown>;
    const source = r.source as Record<string, unknown> | undefined;
    const votes = r.votes as Record<string, unknown> | undefined;
    const currencies = r.currencies as Array<Record<string, unknown>> | undefined;

    if (
      typeof r.id !== 'number' ||
      typeof r.title !== 'string' ||
      typeof r.url !== 'string' ||
      typeof r.published_at !== 'string' ||
      typeof source?.title !== 'string' ||
      typeof votes?.positive !== 'number' ||
      typeof votes?.negative !== 'number' ||
      typeof votes?.important !== 'number' ||
      !Array.isArray(currencies)
    ) {
      return null;
    }

    posts.push({
      id: r.id,
      title: r.title,
      url: r.url,
      sourceTitle: source.title,
      publishedAt: r.published_at,
      currencies: currencies.map((c) => String(c.code)),
      votes: { positive: votes.positive, negative: votes.negative, important: votes.important },
    });
  }
  return posts;
}

export function createNewsService(deps: NewsServiceDeps) {
  const fetchImpl = deps.fetchImpl ?? fetch;
  const now = deps.now ?? Date.now;
  const cacheTtlMs = deps.cacheTtlMs ?? DEFAULT_CACHE_TTL_MS;
  const cache = new Map<string, { expiresAt: number; data: CryptoPanicPost[] }>();

  function cacheKey(query: NewsQuery): string {
    return `${query.currencies ?? ''}|${query.filter ?? ''}`;
  }

  async function fetchNews(query: NewsQuery): Promise<CryptoPanicPost[]> {
    const key = cacheKey(query);
    const cached = cache.get(key);
    if (cached && cached.expiresAt > now()) {
      return cached.data;
    }

    const url = buildRequestUrl(query, deps.apiKey);
    let response: Response;
    try {
      response = await fetchImpl(url);
    } catch (err) {
      throw new UpstreamUnavailableError('CryptoPanic request failed', { cause: err });
    }

    if (response.status === 401 || response.status === 429) {
      throw new UpstreamRejectedError(`CryptoPanic rejected the request (status ${response.status})`);
    }
    if (!response.ok) {
      throw new UpstreamUnavailableError(`CryptoPanic returned status ${response.status}`);
    }

    let payload: unknown;
    try {
      payload = await response.json();
    } catch (err) {
      throw new UpstreamMalformedError('CryptoPanic response was not valid JSON', { cause: err });
    }

    const posts = parsePosts(payload);
    if (posts === null) {
      throw new UpstreamMalformedError('CryptoPanic response missing expected fields');
    }

    cache.set(key, { expiresAt: now() + cacheTtlMs, data: posts });
    return posts;
  }

  return { fetchNews };
}
```

```typescript
// backend/src/routes/news.ts
import type { FastifyInstance } from 'fastify';
import {
  createNewsService,
  UpstreamMalformedError,
  UpstreamRejectedError,
  UpstreamUnavailableError,
} from '../services/cryptopanic.js';

export function registerNewsRoutes(app: FastifyInstance, deps: { apiKey?: string }): void {
  const newsService = deps.apiKey ? createNewsService({ apiKey: deps.apiKey }) : null;

  app.get('/api/news', async (request, reply) => {
    if (!newsService) {
      reply.code(500);
      return { error: 'Server is missing CRYPTOPANIC_API_KEY.' };
    }

    const { currencies, filter } = request.query as { currencies?: string; filter?: string };

    try {
      const posts = await newsService.fetchNews({ currencies, filter });
      return { posts };
    } catch (err) {
      if (err instanceof UpstreamUnavailableError) {
        reply.code(503);
        return { error: 'News provider is unavailable right now.' };
      }
      if (err instanceof UpstreamRejectedError || err instanceof UpstreamMalformedError) {
        reply.code(502);
        return { error: 'News provider returned an unexpected response.' };
      }
      throw err;
    }
  });
}
```

```typescript
// frontend/src/lib/api/news.ts
import type { NewsFilter, NewsPost } from '../../types/news';

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL ?? 'http://localhost:4000';

export async function fetchNews(params: { currencies?: string; filter?: NewsFilter }): Promise<NewsPost[]> {
  const url = new URL('/api/news', API_BASE_URL);
  if (params.currencies) url.searchParams.set('currencies', params.currencies);
  if (params.filter) url.searchParams.set('filter', params.filter);

  const response = await fetch(url.toString());
  const body = await response.json();

  if (!response.ok) {
    throw new Error(typeof body?.error === 'string' ? body.error : 'Failed to load news.');
  }

  return body.posts as NewsPost[];
}
```

```tsx
// frontend/src/components/NewsList.tsx
import type { NewsPost } from '../types/news';

interface NewsListProps {
  status: 'loading' | 'error' | 'ready';
  posts: NewsPost[];
  errorMessage?: string;
  onRetry: () => void;
}

function timeAgo(iso: string): string {
  const diffMs = Date.now() - new Date(iso).getTime();
  const minutes = Math.floor(diffMs / 60_000);
  if (minutes < 1) return 'just now';
  if (minutes < 60) return `${minutes}m ago`;
  const hours = Math.floor(minutes / 60);
  if (hours < 24) return `${hours}h ago`;
  return `${Math.floor(hours / 24)}d ago`;
}

export function NewsList({ status, posts, errorMessage, onRetry }: NewsListProps) {
  if (status === 'loading') {
    return (
      <div className="space-y-3">
        {[0, 1, 2].map((i) => (
          <div key={i} className="h-16 animate-pulse rounded-lg border" style={{ borderColor: 'var(--border)', background: 'var(--surface-1)' }} />
        ))}
      </div>
    );
  }

  if (status === 'error') {
    return (
      <div className="rounded-lg border p-4 text-sm" style={{ borderColor: 'var(--border)', background: 'var(--surface-1)' }}>
        <p style={{ color: 'var(--status-critical)' }}>{errorMessage ?? 'Failed to load news.'}</p>
        <button
          type="button"
          onClick={onRetry}
          className="mt-2 rounded border px-3 py-1 text-xs font-medium"
          style={{ borderColor: 'var(--border)', color: 'var(--text-primary)' }}
        >
          Retry
        </button>
      </div>
    );
  }

  if (posts.length === 0) {
    return (
      <div className="rounded-lg border p-4 text-sm" style={{ borderColor: 'var(--border)', background: 'var(--surface-1)', color: 'var(--text-muted)' }}>
        No news matches the current filter.
      </div>
    );
  }

  return (
    <ul className="space-y-3">
      {posts.map((post) => (
        <li key={post.id} className="rounded-lg border p-4" style={{ borderColor: 'var(--border)', background: 'var(--surface-1)' }}>
          <a href={post.url} target="_blank" rel="noreferrer" className="font-medium hover:underline" style={{ color: 'var(--text-primary)' }}>
            {post.title}
          </a>
          <div className="mt-1 flex flex-wrap items-center gap-2 text-xs" style={{ color: 'var(--text-muted)' }}>
            <span>{post.sourceTitle}</span>
            <span>&middot;</span>
            <span>{timeAgo(post.publishedAt)}</span>
            {post.currencies.map((code) => (
              <span key={code} className="rounded-full border px-2 py-0.5" style={{ borderColor: 'var(--border)' }}>
                {code}
              </span>
            ))}
            {post.votes.positive > post.votes.negative ? (
              <span style={{ color: 'var(--status-good)' }}>▲ {post.votes.positive}</span>
            ) : post.votes.negative > post.votes.positive ? (
              <span style={{ color: 'var(--status-critical)' }}>▼ {post.votes.negative}</span>
            ) : null}
            {post.votes.important > 0 ? <span style={{ color: 'var(--status-warning)' }}>★ important</span> : null}
          </div>
        </li>
      ))}
    </ul>
  );
}
```

```tsx
// frontend/src/pages/NewsPage.tsx
import { useEffect, useState } from 'react';
import { NewsFilters } from '../components/NewsFilters';
import { NewsList } from '../components/NewsList';
import { fetchNews } from '../lib/api/news';
import type { NewsFilter, NewsPost } from '../types/news';

export function NewsPage() {
  const [currencies, setCurrencies] = useState('');
  const [filter, setFilter] = useState<NewsFilter | undefined>(undefined);
  const [status, setStatus] = useState<'loading' | 'error' | 'ready'>('loading');
  const [posts, setPosts] = useState<NewsPost[]>([]);
  const [errorMessage, setErrorMessage] = useState<string | undefined>(undefined);
  const [requestId, setRequestId] = useState(0);

  useEffect(() => {
    let cancelled = false;
    setStatus('loading');

    fetchNews({ currencies: currencies || undefined, filter })
      .then((result) => {
        if (cancelled) return;
        setPosts(result);
        setStatus('ready');
      })
      .catch((err: unknown) => {
        if (cancelled) return;
        setErrorMessage(err instanceof Error ? err.message : 'Failed to load news.');
        setStatus('error');
      });

    return () => {
      cancelled = true;
    };
  }, [currencies, filter, requestId]);

  return (
    <div className="space-y-4">
      <h1 className="text-xl font-semibold" style={{ color: 'var(--text-primary)' }}>
        Crypto News
      </h1>
      <NewsFilters currencies={currencies} onCurrenciesChange={setCurrencies} filter={filter} onFilterChange={setFilter} />
      <NewsList status={status} posts={posts} errorMessage={errorMessage} onRetry={() => setRequestId((n) => n + 1)} />
    </div>
  );
}
```
